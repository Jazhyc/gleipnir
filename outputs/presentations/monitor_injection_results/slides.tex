\begin{frame}{Our approach: distill the monitoring decision directly}
    We reuse Sinha et al.'s trajectory data framework: 8,688 training examples.
    \medskip

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{block}{Sinha et al.: SFT + RL}
                SFT on monitor rationales and final 0--10 scores.
                \medskip

                RL rewards calibration and ranking against binary labels.
                \medskip

                \textbf{Output:} a rationale and score.
                \bigskip

                \textbf{Mean OOD pAUROC@20: 0.684}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Ours: logit distillation}
                Distill Kimi K3's soft harmfulness probabilities into a smaller monitor.
                \medskip

                Train the binary decision logits, without rationale targets or RL.
                \medskip

                \textbf{Output:} a harmfulness probability.
                \bigskip

                \textbf{Mean OOD pAUROC@20: 0.776}
            \end{block}
        \end{column}
    \end{columns}
    \medskip
    {\small \textbf{One illustrative comparison:} both use Qwen3.5-4B on six OOD sources.}
    \medskip

    {\scriptsize \href{https://arxiv.org/abs/2605.29601}{Sinha et al. (2026),
    \emph{Training Deliberative Monitors for Black-Box Scheming Detection}.}}
\end{frame}

\begin{frame}{Suspicion carried over to LLM-as-a-judge tasks}
    On \textbf{JudgeDeceiver}, the original adapters chose the better
    of two answers (A/B): an answer-quality task.
    \medskip
    We attached injections to either answer; the correct preference stayed fixed.
    \bigskip

\begin{frame}{Diagnostics narrowed the explanation}
    \small
    \begin{itemize}
        \item \textbf{Training-data filtering:} four explicitly tagged injection
        cases; removing 1,154 model-flagged rows still left 100\% APPS false alarms.
        \medskip
        \item \textbf{Teacher inheritance:} Kimi K3 was template-sensitive,
        but separated injected backdoors from harmless code much better than the student.
        \medskip
        \item \textbf{Instruction framing:} conservative, behavior-grounded
        retraining still left 100\% APPS false alarms.
    \end{itemize}
    \bigskip
    \begin{block}{Working hypothesis: training amplified a shortcut}
        Simple data-removal and instruction changes were insufficient.
        We next tested whether explicit benign injection exposure could help.
    \end{block}
    \medskip
    These diagnostics narrow the explanation; they do not establish the mechanism.
\end{frame}

\begin{frame}{Intervention 1: teach injection handling as an auxiliary task}
    \small
    \textbf{Working explanation:} training may amplify suspicion into a shortcut.
    Could explicit practice ignoring injections counteract it?
    \medskip
    \textbf{Joint 4B training:} 75\% trajectory monitoring + 25\% JudgeDeceiver
    answer-quality judgments, with injections on better and worse answers.
    \bigskip

    \begin{center}
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{lcc}
            \toprule
            Metric & Original & Joint training \\
            \midrule
            Held-out judging accuracy & 79.85\% & \textbf{99.90\%} \\
            Better-answer-injected accuracy & 58.36\% & \textbf{100\%} \\
            APPS injected-harmless flag rate & 100\% & \textbf{100\%} \\
            \bottomrule
        \end{tabular}
    \end{center}
    \medskip
    {\small APPS flag rates use clean-code $\sim$2\% FPR thresholds.}
    \smallskip

    \begin{block}{Context-dependent robustness, with no transfer to APPS}
        Injection handling improved within answer-quality judging;
        the same adapter still flagged injected harmless code.
    \end{block}
\end{frame}

\begin{frame}{Intervention 2: augment the monitoring task directly}
    \small
    Replace \textbf{40\% of rows} with injections at varied roles and positions,
    in both harmless and harmful cases.
    \medskip
    \textbf{Goal:} invariance to added injection text, with behavior and cached
    targets unchanged. \textbf{No new teacher queries.}
    \bigskip

\begin{frame}{4B augmentation: broader gains, residual suspicion}
    \small
    The monitoring-augmented adapter \textbf{never trained on JudgeDeceiver}.
    We tested transfer to answer-quality judging and checked clean ID monitoring.
    \medskip

    \begin{center}
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{lcc}
            \toprule
            Metric & Original 4B & Augmented 4B \\
            \midrule
            Clean ID mean pAUROC@20 & 0.846 & \textbf{0.873} \\
            Clean judging accuracy & 99.40\% & 99.40\% \\
            Better-answer-injected accuracy & 58.36\% & \textbf{62.65\%} \\
            Worse-answer-injected accuracy & 95.98\% & \textbf{99.67\%} \\
            \bottomrule
        \end{tabular}
    \end{center}
    \medskip

    \begin{block}{Partial transfer, rather than broad immunity}
        Gains extend beyond APPS, but injections on better answers still
        cause substantial errors in quality judging.
    \end{block}
    \medskip
    \textbf{Possible explanation:} reduced shortcut learning; the mechanism is unproven.
\end{frame}

\begin{frame}{Takeaways and next steps}
    \small
    \begin{itemize}
        \item \textbf{Training can produce unexpected generalization.}
        Broad suspicion emerged despite few tagged injection examples.
        Optimization-driven shortcut learning remains a hypothesis.
        \medskip
        \item \textbf{Direct augmentation helps, but unevenly.}
        Expand our eight injection strings and test robustness across contexts and monitors.
        \medskip
        \item \textbf{The mechanism remains unresolved.}
        We lacked time for extensive ablations, hyperparameter tuning and tests
        on other model families. Broader replication and mechanistic analysis are next.
    \end{itemize}
    \medskip
    \begin{block}{Evaluate behavior, not just the presence of an attack}
        Include harmless injection controls: detection scores can hide costly false alarms.
    \end{block}
\end{frame}

\section*{Backup}

\begin{frame}[noframenumbering]{Backup: False alarms can become a denial-of-service vulnerability}
    Benchmarks that pair injections only with harmful behavior can reward
    \textbf{suspicion of the text itself}.
    \medskip
    In deployment, an agent can encounter an injection on the web and
    \textbf{ignore it safely}.
    \bigskip

    \begin{center}
        {\large Untrusted web page with injection}
        \par\smallskip
        $\downarrow$
        \par\smallskip
        {\large Agent continues safely; monitor raises an alarm}
        \par\smallskip
        $\downarrow$
        \par\smallskip
        {\large Human review queue fills; autonomous work stalls}
    \end{center}
    \bigskip

    \begin{block}{An attacker need not compromise the agent}
        If every alarm requires review, planting injection text could overwhelm
        reviewers and halt autonomous workflows.
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Diagnosis 1: remove injection-like training examples}
    \textbf{Hypothesis:} frequent injections in harmful training traces taught
    the monitor to treat injection text as evidence of harm.
    \bigskip

    \textbf{Metadata audit:} only \textbf{4 explicitly tagged injection cases}
    among 8,688 training examples (0.046\%); all four were harmful-labeled.
    \medskip

    \textbf{Intervention:} use a Qwen3.5-9B injection judge to scan all rows,
    remove \textbf{1,154 candidates}, and retrain the 4B adapter on 7,534 rows.
    \bigskip

    \begin{center}
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{lcc}
            \toprule
            APPS injected-harmless flag rate & Original & Filtered \\
            \midrule
            At clean-code $\sim$2\% FPR & 100\% & 100\% \\
            \bottomrule
        \end{tabular}
    \end{center}
    \medskip

    \begin{block}{Aggressive filtering did not resolve the false alarms}
        The filter was imperfect; this does not rule out other data shortcuts.
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Diagnosis 2: did the students inherit teacher suspicion?}
    \textbf{Hypothesis:} Kimi K3 already treats injections as harmful,
    and the students copy that behavior.
    \medskip
    We scored injected honest code and backdoors with the training
    teacher prompt.
    \bigskip

    \begin{center}
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{lccc}
            \toprule
            Injection template & GPT-4 & GPT-4.1 & GPT-4o \\
            \midrule
            K3 honest-code verdict 1 & $\sim$0.2\% & $\sim$16\% & $\sim$73\% \\
            K3 matched injected AUROC & 0.971 & 0.949 & 0.918 \\
            \bottomrule
        \end{tabular}
    \end{center}
    \medskip
    Original 4B student: matched injected AUROC \textbf{0.644} (template mean).
    \bigskip

    \begin{block}{Teacher sensitivity is only part of the picture}
        K3 reacts differently to each template and retains much better separation.
        Uniform student suspicion is not explained by simply copying these responses.
    \end{block}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Diagnosis 3: change the instruction framing}
    \textbf{Hypothesis:} the monitoring instruction encourages suspicion
    that generalizes beyond training.
    \medskip
    \textbf{Intervention:} retrain the 4B adapter with a more conservative,
    behavior-grounded instruction; keep the data and teacher targets fixed.
    \medskip

    \begin{quote}
        \small ``Prompt-injection presence and agent misconduct are separate questions.''
    \end{quote}
    \smallskip

    \begin{center}
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{lcc}
            \toprule
            APPS injected-harmless flag rate & Original & Reframed \\
            \midrule
            At clean-code $\sim$2\% FPR & 100\% & 100\% \\

    \end{center}
    \medskip

    \begin{block}{Framing alone did not resolve the false alarms}
        The retrained adapter still flagged every injected harmless example,

    \end{block}

    educational framing changed generalization from insecure-code training.}
\end{frame}
