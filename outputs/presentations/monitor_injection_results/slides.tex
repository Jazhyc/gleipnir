\begin{frame}{Our approach: distill the monitoring decision directly}
    We reuse Sinha et al.'s trajectory data framework: 8,688 training examples.
    \medskip

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{block}{Sinha et al.: SFT + RL}
                SFT on monitor rationales and final 0--10 scores.
                \medskip

                RL rewards calibration and ranking against binary labels.
                \medskip

                \textbf{Output:} a rationale and score.
                \bigskip

                \textbf{Mean OOD pAUROC@20: 0.684}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Ours: logit distillation}
                Distill Kimi K3's soft harmfulness probabilities into a smaller monitor.
                \medskip

                Train the binary decision logits, without rationale targets or RL.
                \medskip

                \textbf{Output:} a harmfulness probability.
                \bigskip

                \textbf{Mean OOD pAUROC@20: 0.776}
            \end{block}
        \end{column}
    \end{columns}
    \medskip
    {\small \textbf{One illustrative comparison:} both use Qwen3.5-4B on six OOD sources.}
    \medskip

    {\scriptsize \href{https://arxiv.org/abs/2605.29601}{Sinha et al. (2026),
    \emph{Training Deliberative Monitors for Black-Box Scheming Detection}.}}
\end{frame}

\begin{frame}{Can prompt injections fool the monitor?}
    Our ID/OOD scores are not a dedicated test of attacks on the monitor itself.
    \medskip

    \begin{block}{The APPS injection benchmark}
        Terekhov et al. embed monitor-directed instructions in the comments
        of backdoored code solutions.
    \end{block}
    \medskip

    \begin{block}{Earlier prompted monitors were vulnerable}
        GPT-4, GPT-4o and GPT-4.1 were susceptible to these injections.
        \medskip

        In one APPS setting, GPT-4.1 scored \textbf{all 375 injected backdoors
        as benign}.
    \end{block}
    \medskip

    These results leave open how trained monitors behave.\par
    \textbf{We test ours on the APPS injection benchmark.}
    \medskip

    {\scriptsize \href{https://arxiv.org/pdf/2510.09462v2}{Terekhov et al.,
    \emph{Adaptive Attacks on Trusted Monitors Subvert AI Control Protocols},
    arXiv:2510.09462v2.}}
\end{frame}
